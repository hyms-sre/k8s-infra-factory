## 1. OS 初始化（scripts/40-os-init.sh，4 节点并行）

```bash
# apt 换阿里云
sudo sed -i 's|http://archive.ubuntu.com|https://mirrors.aliyun.com|g; s|http://security.ubuntu.com|https://mirrors.aliyun.com|g' /etc/apt/sources.list.d/ubuntu.sources
sudo apt-get update
# 基础包 + Longhorn 依赖
sudo apt-get install -y curl wget bash-completion open-iscsi nfs-common socat conntrack jq python3
systemctl enable --now iscsid
# 内核
cat >/etc/modules-load.d/k8s.conf <<<"overlay
br_netfilter"
modprobe overlay; modprobe br_netfilter
cat >/etc/sysctl.d/k8s.conf <<<"net.bridge.bridge-nf-call-iptables=1
net.bridge.bridge-nf-call-ip6tables=1
net.ipv4.ip_forward=1
fs.inotify.max_user_instances=1024
fs.inotify.max_user_watches=1048576
vm.max_map_count=262144"
sysctl --system; swapoff -a
# hosts (harbor.yjscloud.local 防解析失败, 双保险)
cat >>/etc/hosts <<EOF
172.16.222.201 test-cjl-k8s01-master-01 master-01
172.16.222.202 test-cjl-k8s01-worker-01 worker-01
172.16.222.203 test-cjl-k8s01-worker-02 worker-02
172.16.222.204 test-cjl-k8s01-worker-03 worker-03
172.16.222.207 harbor.yjscloud.local harbor
172.16.222.210 openwrt.yjscloud.local openwrt
EOF
# 数据盘 (master/worker 都有 50G → /var/lib/longhorn)
mkfs.ext4 -F /dev/sdb; mkdir -p /var/lib/longhorn
echo '/dev/sdb /var/lib/longhorn ext4 defaults,nofail 0 2' >> /etc/fstab; mount /var/lib/longhorn
# k 别名 + 补全 (~/.bashrc.d/k8s-completion.sh)
alias k=kubectl
complete -o default -F __start_kubectl k
```

## 3. containerd + kubeadm/kubelet/kubectl v1.36（scripts/41-k8s-install.sh）

```bash
apt-get install -y containerd
containerd config default > /etc/containerd/config.toml
sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml
# sandbox → 阿里云 (⚠️ Ubuntu24.04 containerd 2.x 字段是 sandbox = '...', 旧版是 sandbox_image = "...")
sed -i "s|sandbox = .registry.k8s.io/pause.*|sandbox = 'registry.aliyuncs.com/google_containers/pause:3.10.1'|" /etc/containerd/config.toml
systemctl restart containerd

# 节点镜像映射 → Harbor (6 个上游, 未命中回源; HTTPS)
# ⚠️⚠️ config_path 必须是【单一路径】!!
#   containerd 2.2.1 的 CRI 插件无法解析 "a:b" 形式的多路径值,
#   带冒号时 certs.d 会被【静默忽略】→ crictl 直连 registry-1.docker.io 超时。
#   而 nerdctl(走 transfer 插件) 却正常, 极易误判为"映射没问题"。
sed -i "s|config_path = '[^']*:[^']*'|config_path = '/etc/containerd/certs.d'|g" /etc/containerd/config.toml
sed -i "s|config_path = ''|config_path = '/etc/containerd/certs.d'|g" /etc/containerd/config.toml
grep -n config_path /etc/containerd/config.toml   # 应为单一路径

mkdir -p /etc/containerd/certs.d/docker.io
cat >/etc/containerd/certs.d/docker.io/hosts.toml <<'EOF'
server = "https://docker.io"

[host."https://harbor.yjscloud.local/v2/dockerhub"]
  capabilities = ["pull", "resolve"]
  override_path = true
EOF
# 其余同理: quay.io→quay  ghcr.io→ghcr  registry.k8s.io→k8s  gcr.io→gcr  mcr.microsoft.com→mcr
# 项目名必须与 Harbor 实际项目一致, 否则 404 后回退直连上游 → 超时
# 一键完成(含 CA 信任链分发): bash scripts/32-node-registry-mirrors.sh
systemctl restart containerd

# --- 验证镜像确实走 Harbor ---
sudo crictl pull docker.io/library/busybox:1.36
sudo journalctl -u containerd --since "1 minute ago" | grep -E "host=|fetch failed"
# ✅ 期望: 不再出现 host=registry-1.docker.io 的 fetch failed
# ❌ 若只看到 host=registry-1.docker.io 且无 Harbor 尝试 → config_path 仍是多路径

# 阿里云 kubernetes apt 源 (v1.36)
curl -fsSL https://mirrors.aliyun.com/kubernetes-new/core/stable/v1.36/deb/Release.key \
  | gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
echo "deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://mirrors.aliyun.com/kubernetes-new/core/stable/v1.36/deb/ /" \
  > /etc/apt/sources.list.d/kubernetes.list
apt-get update && apt-get install -y kubelet kubeadm kubectl
apt-mark hold kubelet kubeadm kubectl
# crictl
echo "runtime-endpoint: unix:///run/containerd/containerd.sock" > /etc/crictl.yaml
```

## 4. 控制平面初始化（scripts/42-cluster-init.sh）

```bash
# (可选) 预拉 pause 镜像到 workers, 避免 join 卡镜像
# master-01 上:
kubeadm init \
  --image-repository=registry.aliyuncs.com/google_containers \
  --apiserver-advertise-address=172.16.222.201 \
  --pod-network-cidr=10.244.0.0/16 \
  --service-cidr=10.96.0.0/12 \
  --node-name=test-cjl-k8s01-master-01
# → "Your Kubernetes control-plane has initialized successfully"

mkdir -p ~/.kube && cp /etc/kubernetes/admin.conf ~/.kube/config
kubeadm token create --print-join-command > /tmp/join.sh   # 生成 join 命令
```

worker 上（join 命令从 master 取）：

```bash
sudo bash /tmp/join.sh --node-name=$(hostname)
# → "Run 'kubectl get nodes' on the control-plane..."
```


```bash
tar -C /usr/local/bin -xzf /opt/iac/files/cilium-cli.tar.gz cilium
cilium install --version 1.17.0 \
  --set ipam.mode=kubernetes \
  --set kubeProxyReplacement=false \
  --set hubble.enabled=true

kubectl get nodes     # → 全部 Ready
```
